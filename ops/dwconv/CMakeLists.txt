if(CUDAToolkit_VERSION VERSION_LESS 11.8)
    message(FATAL_ERROR "DWConv 的 cudaLaunchKernelEx 启动接口需要 CUDA Toolkit 11.8 或更新版本")
endif()

cuda_lab_add_operator(dwconv SOURCES dwconv_pytorch.cu dwconv_fp32.cu dwconv_fp16.cu dwconv_opt_fp32.cu dwconv_opt_fp16.cu)

# 旧构建目录沿用原开关值；显式指定新开关时优先使用新值。
if(DEFINED DWCONV_TEST_CUDNN AND NOT DEFINED DWCONV_ENABLE_CUDNN)
    set(DWCONV_ENABLE_CUDNN "${DWCONV_TEST_CUDNN}" CACHE BOOL "构建 DWConv cuDNN 后端")
endif()
option(DWCONV_ENABLE_CUDNN "构建 DWConv cuDNN 后端" ON)
if(DWCONV_ENABLE_CUDNN)
    find_path(CUDNN_INCLUDE_DIR cudnn.h HINTS ${CUDAToolkit_INCLUDE_DIRS})
    find_library(CUDNN_LIBRARY cudnn HINTS ${CUDAToolkit_LIBRARY_DIR})
    if(CUDNN_INCLUDE_DIR AND CUDNN_LIBRARY)
        cuda_lab_add_operator(dwconv_cudnn SOURCES dwconv_cudnn.cu)
        target_include_directories(cuda_ops_dwconv_cudnn PRIVATE ${CUDNN_INCLUDE_DIR})
        target_link_libraries(cuda_ops_dwconv_cudnn PRIVATE ${CUDNN_LIBRARY})
    else()
        message(STATUS "未找到 cuDNN，跳过 DWConv cuDNN 后端")
    endif()
endif()

cuda_lab_add_operator(dwconv_pytorch SOURCES dwconv_pytorch_dispatch.cpp dwconv_pytorch_dispatch.cu)
target_link_libraries(cuda_ops_dwconv_pytorch PUBLIC cuda_lab::dwconv)
if(TARGET cuda_ops_dwconv_cudnn)
    target_compile_definitions(cuda_ops_dwconv_pytorch PRIVATE DWCONV_PYTORCH_HAS_CUDNN=1)
    target_include_directories(cuda_ops_dwconv_pytorch PRIVATE ${CUDNN_INCLUDE_DIR})
    target_link_libraries(cuda_ops_dwconv_pytorch PRIVATE cuda_lab::dwconv_cudnn)
endif()

if(DEFINED DWCONV_TEST_TENSORRT AND NOT DEFINED DWCONV_ENABLE_TENSORRT)
    set(DWCONV_ENABLE_TENSORRT "${DWCONV_TEST_TENSORRT}" CACHE BOOL "构建 DWConv TensorRT 后端（需要 TensorRT 10）")
endif()
option(DWCONV_ENABLE_TENSORRT "构建 DWConv TensorRT 后端（需要 TensorRT 10）" ON)
if(DWCONV_ENABLE_TENSORRT)
    find_path(TENSORRT_INCLUDE_DIR NvInfer.h PATH_SUFFIXES x86_64-linux-gnu)
    find_library(TENSORRT_LIBRARY nvinfer)
    if(TENSORRT_INCLUDE_DIR AND TENSORRT_LIBRARY)
        cuda_lab_add_operator(dwconv_tensorrt SOURCES dwconv_tensorrt.cu dwconv_tensorrt_plugin.cu)
        target_include_directories(cuda_ops_dwconv_tensorrt PRIVATE ${TENSORRT_INCLUDE_DIR})
        target_link_libraries(cuda_ops_dwconv_tensorrt PRIVATE cuda_lab::dwconv ${TENSORRT_LIBRARY})
    else()
        message(STATUS "未找到 TensorRT，跳过 DWConv TensorRT 后端")
    endif()
endif()
